# Пример настроек media-monitor.
# Рабочий файл — settings.txt (создаётся из этого примера один раз, потом не перезаписывается).
# Этот файл settings.example.txt можно обновлять при новых версиях — это только образец.

# Дата ДО (включительно). Формат: ГГГГ-ММ-ДД.
# Берутся статьи с датой выхода не позже этой.
# Если пусто — используется сегодняшняя дата.
article_date_not_later_than=

# Дата НЕ СТАРШЕ (включительно). Формат: ГГГГ-ММ-ДД.
# Берутся статьи с датой выхода не раньше этой.
# Если пусто — нижней границы нет.
# Если у статьи дата не найдена — статья всё равно попадает в отчёт.
article_date_not_older_than=

# Сколько секунд ждать ввод логина/пароля при старте (для сайтов с авторизацией).
# Пустой логин или таймаут = работать без авторизации.
auth_timeout_seconds=180

# Максимум страниц на ОДИН сайт за запуск.
# 0 = без ограничения.
max_scan_urls=0

# Максимум ссылок, которые берутся с одной страницы списка/поиска.
max_expand_links=120

# Сколько сайтов сканировать одновременно (каждый сайт — свой поток).
# Внутри сайта запросы всегда идут по одному.
# 0 = все сайты сразу (рекомендуется).
# Например 8 — не больше 8 сайтов параллельно.
site_workers=0

# Подробный лог: 0 = только прогресс сайтов и HIT (ошибки скрыты),
# 1 = каждый URL, SSL, warn/error.
log_verbose=0

# Если сайт полностью недоступен — закомментировать его строку в sites.txt (#).
# 1 = да, 0 = нет. В Excel недоступные сайты всё равно пишутся в конце отчёта (красным).
comment_unavailable_sites=1

# Проверять SSL-сертификаты сайтов: 1 = да, 0 = нет.
# При 1 — встроенный набор CA (certifi + корни Минцифры + нужные intermediate).
# Отдельно ставить сертификаты в Windows не нужно.
# Если сертификат всё равно битый — сайт запоминается и дальше без проверки.
# ssl_verify=0 — никогда не проверять (чуть быстрее на совсем «кривых» сайтах).
ssl_verify=1
